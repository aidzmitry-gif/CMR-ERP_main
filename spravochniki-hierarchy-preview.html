Как это устроено в БД
-- Обход поддерева «Электронные компоненты» с защитой от циклов
WITH RECURSIVE subtree AS (
    SELECT id, parent_id, name, code, 1 AS level
    FROM public.nomenclature_category
    WHERE code = 'CAT-0100'                 -- корень поддерева
  UNION ALL
    SELECT c.id, c.parent_id, c.name, c.code, s.level + 1
    FROM public.nomenclature_category c
    JOIN subtree s ON c.parent_id = s.id
) CYCLE id SET is_cycle USING path        -- стоп при повторе узла (цикл)
SELECT code, name, level FROM subtree
WHERE NOT is_cycle
ORDER BY level, name;

-- Производный ltree-путь для быстрых поддеревьев и AI (только PostgreSQL):
-- SELECT * FROM nomenclature_category WHERE ltree_path <@ 'catalog.electronics';

parent_id — источник истины иерархии (переносим в SQLite-dev, рекурсивный CTE с защитой от циклов через CYCLE). ltree-путь — производный индекс для быстрых выборок поддеревьев и структурного доступа AI (GiST-индекс, только PostgreSQL).
